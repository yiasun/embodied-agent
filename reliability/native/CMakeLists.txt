cmake_minimum_required(VERSION 3.16)
project(ea_bounded_motion LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(ea_bounded_motion SHARED BoundedMotion.cc)
target_compile_features(ea_bounded_motion PRIVATE cxx_std_17)
target_compile_options(ea_bounded_motion PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ea_bounded_motion PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
